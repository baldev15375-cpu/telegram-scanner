import os, time, requests
import pandas as pd
import telebot

TOKEN = os.getenv("BOT_TOKEN") or os.getenv("TELEGRAM_TOKEN")
bot = telebot.TeleBot(TOKEN)
CHAT_ID_FILE = "chat_id.txt"

def get_chat_id():
    if os.path.exists(CHAT_ID_FILE):
        return open(CHAT_ID_FILE).read().strip()
    return None

@bot.message_handler(commands=['start'])
def start(m):
    open(CHAT_ID_FILE,'w').write(str(m.chat.id))
    bot.reply_to(m, "✅ Futures 50/200 4H + ADX Scanner Started - Binance Top 350\nAlert every 10 min")

def get_futures_top_symbols():
    url = "https://fapi.binance.com/fapi/v1/ticker/24hr"
    data = requests.get(url).json()
    usdt = [d for d in data if d['symbol'].endswith('USDT') and 'PERP' not in d['symbol']]
    usdt_sorted = sorted(usdt, key=lambda x: float(x['quoteVolume']), reverse=True)[:350]
    return [d['symbol'] for d in usdt_sorted]

def get_klines(symbol, interval="4h", limit=250):
    url = f"https://fapi.binance.com/fapi/v1/klines?symbol={symbol}&interval={interval}&limit={limit}"
    r = requests.get(url).json()
    df = pd.DataFrame(r, columns=['o','h','l','c','v','ct','q','n','tb','tq','i'])
    df['c'] = df['c'].astype(float)
    return df

def check_golden_cross_adx(df):
    df['ema50'] = df['c'].ewm(span=50).mean()
    df['ema200'] = df['c'].ewm(span=200).mean()
    # ADX simple calculation
    df['tr'] = df['h'].astype(float) - df['l'].astype(float)
    # simplified ADX > 20 trend filter using ema slope
    if len(df) < 210: return False
    prev_50 = df['ema50'].iloc[-2]
    prev_200 = df['ema200'].iloc[-2]
    curr_50 = df['ema50'].iloc[-1]
    curr_200 = df['ema200'].iloc[-1]
    # golden cross just happened and ADX trend strong (ema distance increasing)
    golden = prev_50 <= prev_200 and curr_50 > curr_200
    adx_filter = (curr_50 - curr_200) > (prev_50 - prev_200) * 1.01
    return golden and adx_filter

def scanner_loop():
    while True:
        try:
            chat_id = get_chat_id()
            if chat_id:
                symbols = get_futures_top_symbols()
                for sym in symbols:
                    try:
                        df = get_klines(sym)
                        if check_golden_cross_adx(df):
                            price = df['c'].iloc[-1]
                            msg = f"🚀 FUTURES GOLDEN CROSS 50/200 4H\nCoin: {sym}\nPrice: {price}\nADX Trend: Strong\nTime: 4H"
                            bot.send_message(chat_id, msg)
                            time.sleep(1)
                    except: continue
            time.sleep(600)
        except Exception as e:
            print(e)
            time.sleep(60)

import threading
threading.Thread(target=scanner_loop, daemon=True).start()
bot.infinity_polling()
