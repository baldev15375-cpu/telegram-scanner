import os
import time
import threading
import pandas as pd
import requests
import telebot
from flask import Flask

TOKEN = os.environ["BOT_TOKEN"]
bot = telebot.TeleBot(TOKEN)
CHAT_ID_FILE = "chat_id.txt"

app = Flask(__name__)
@app.route('/')
def home():
    return "Bot Running"

def run_flask():
    app.run(host='0.0.0.0', port=8080)

def get_chat_id():
    if os.path.exists(CHAT_ID_FILE):
        with open(CHAT_ID_FILE, encoding="utf-8") as f:
            return f.read().strip()
    return None

@bot.message_handler(commands=["start"])
def start(message):
    with open(CHAT_ID_FILE, "w", encoding="utf-8") as f:
        f.write(str(message.chat.id))
    bot.reply_to(message, "✅ Scanner Started 24/7")

@bot.message_handler(commands=["status"])
def status(message):
    bot.reply_to(message, "Bot is Running 24/7 - Futures 50/200")

def get_futures_top_symbols():
    url = "https://fapi.binance.com/fapi/v1/ticker/24hr"
    r = requests.get(url, timeout=20)
    r.raise_for_status()
    data = r.json()
    usdt = [i for i in data if i["symbol"].endswith("USDT") and "PERP" not in i["symbol"]]
    usdt = sorted(usdt, key=lambda x: float(x["quoteVolume"]), reverse=True)[:350]
    return [i["symbol"] for i in usdt]

def get_klines(symbol, interval="4h", limit=250):
    url = "https://fapi.binance.com/fapi/v1/klines"
    r = requests.get(url, params={"symbol": symbol, "interval": interval, "limit": limit}, timeout=20)
    r.raise_for_status()
    data = r.json()
    cols = ["open_time","o","h","l","c","v","close_time","q","n","tb","tq","ignore"]
    df = pd.DataFrame(data, columns=cols)
    for c in ("h","l","c"):
        df[c] = df[c].astype(float)
    return df

def check_golden_cross_adx(df):
    df = df.copy()
    df["ema50"] = df["c"].ewm(span=50).mean()
    df["ema200"] = df["c"].ewm(span=200).mean()
    if len(df) < 210:
        return False
    prev_50 = df["ema50"].iloc[-2]
    prev_200 = df["ema200"].iloc[-2]
    curr_50 = df["ema50"].iloc[-1]
    curr_200 = df["ema200"].iloc[-1]
    golden = prev_50 <= prev_200 and curr_50 > curr_200
    adx_filter = (curr_50 - curr_200) > (prev_50 - prev_200) * 1.01
    return golden and adx_filter

def scanner_loop():
    while True:
        try:
            chat_id = get_chat_id()
            if chat_id:
                for symbol in get_futures_top_symbols():
                    try:
                        df = get_klines(symbol)
                        if check_golden_cross_adx(df):
                            price = df["c"].iloc[-1]
                            msg = f"🚀 GOLDEN CROSS 50/200 4H\nCoin: {symbol}\nPrice: {price}"
                            bot.send_message(chat_id, msg)
                            time.sleep(1)
                    except Exception as e:
                        print(e)
            time.sleep(600)
        except Exception as e:
            print(e)
            time.sleep(60)

if __name__ == "__main__":
    threading.Thread(target=run_flask, daemon=True).start()
    threading.Thread(target=scanner_loop, daemon=True).start()
    bot.infinity_polling()
